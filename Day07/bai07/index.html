function findMaxNumber(a, b, c) {
  if (isNaN(a) || isNaN(b) || isNaN(c)) {
    return "a hoặc b hoặc c không hợp lệ";
  }

  let max = a;
  if (max < b) {
    max = b;
  }

  if (max < c) {
    max = c;
  }

  return max;
}
